      JavaScript (Yoast JSON-LD Graph Verifier)
      
export function validateYoastGraph(jsonLd) {
  const graph = jsonLd['@graph'] || [];
  const definedIds = new Set(graph.map(node => node['@id']).filter(Boolean));
  const brokenRefs = [];
  for (const node of graph) {
    for (const [k, v] of Object.entries(node)) {
      if (v && typeof v === 'object' && v['@id'] && !definedIds.has(v['@id'])) {
        brokenRefs.push({ from: node['@type'], property: k, missingId: v['@id'] });
      }
    }
  }
  return { nodeCount: graph.length, definedIds: definedIds.size, brokenRefs };
}
